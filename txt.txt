// Carousel confined in hero-visual and FAQ accordion behaviour
// - Carousel auto-plays and supports prev/next and indicators
// - FAQ items toggle on click/Enter and maintain accessibility
// - Responsive-friendly: ensures carousel works on mobile (controls visible, keyboard support)

(function () {
  // select hero visual and slides
  const heroVisual = document.getElementById('hero-visual');
  if (heroVisual) {
    const slides = Array.from(heroVisual.querySelectorAll('.image'));
    const prevBtn = document.getElementById('carousel-prev');
    const nextBtn = document.getElementById('carousel-next');
    const indicatorsWrap = document.getElementById('carousel-indicators');

    let idx = 0;
    let autoplayTimer = null;
    const AUTOPLAY_DELAY = 4200; // ms

    // defensive: if no slides, hide controls
    if (!slides.length) {
      if (indicatorsWrap) indicatorsWrap.style.display = 'none';
      if (prevBtn) prevBtn.style.display = 'none';
      if (nextBtn) nextBtn.style.display = 'none';
    } else {
      // build indicators
      function createIndicators() {
        if (!indicatorsWrap) return;
        indicatorsWrap.innerHTML = '';
        slides.forEach((s, i) => {
          const b = document.createElement('button');
          b.type = 'button';
          b.setAttribute('aria-label', 'Aller à l’image ' + (i + 1));
          b.className = i === 0 ? 'active' : '';
          b.addEventListener('click', () => {
            goTo(i);
            resetAutoplay();
          });
          indicatorsWrap.appendChild(b);
        });
      }

      function showIndex(n) {
        slides.forEach((s, i) => s.classList.toggle('active', i === n));
        if (indicatorsWrap) {
          Array.from(indicatorsWrap.children).forEach((b, i) => b.classList.toggle('active', i === n));
        }
      }

      function prev() { idx = (idx - 1 + slides.length) % slides.length; showIndex(idx); }
      function next() { idx = (idx + 1) % slides.length; showIndex(idx); }

      function goTo(n) { idx = ((n % slides.length) + slides.length) % slides.length; showIndex(idx); }

      function startAutoplay() {
        stopAutoplay();
        autoplayTimer = setInterval(() => next(), AUTOPLAY_DELAY);
      }
      function stopAutoplay() { if (autoplayTimer) { clearInterval(autoplayTimer); autoplayTimer = null; } }
      function resetAutoplay() { stopAutoplay(); startAutoplay(); }

      // events
      if (prevBtn) prevBtn.addEventListener('click', () => { prev(); resetAutoplay(); });
      if (nextBtn) nextBtn.addEventListener('click', () => { next(); resetAutoplay(); });

      // keyboard nav: left/right when heroVisual focused
      heroVisual.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowLeft') { prev(); resetAutoplay(); }
        else if (e.key === 'ArrowRight') { next(); resetAutoplay(); }
      });

      // start
      createIndicators();
      showIndex(0);
      startAutoplay();

      // pause/resume on hover/focus for accessibility
      heroVisual.addEventListener('mouseenter', stopAutoplay);
      heroVisual.addEventListener('mouseleave', startAutoplay);
      heroVisual.addEventListener('focusin', stopAutoplay);
      heroVisual.addEventListener('focusout', startAutoplay);
    }
  }

  // FAQ accordion (accessible)
  const faqItems = document.querySelectorAll('.faq-item');
  faqItems.forEach(item => {
    const q = item.querySelector('.faq-question');
    const a = item.querySelector('.faq-answer');
    if (!q || !a) return;

    q.tabIndex = 0;
    q.setAttribute('role', 'button');
    q.setAttribute('aria-expanded', 'false');
    a.setAttribute('aria-hidden', 'true');

    function toggle() {
      const isOpen = item.classList.toggle('open');
      q.setAttribute('aria-expanded', isOpen ? 'true' : 'false');
      a.setAttribute('aria-hidden', isOpen ? 'false' : 'true');
    }

    q.addEventListener('click', toggle);
    q.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); toggle(); }
    });
  });

  // small entrance animations for cards
  document.addEventListener('DOMContentLoaded', () => {
    document.querySelectorAll('.card').forEach((el, i) => {
      el.style.opacity = 0;
      el.style.transform = 'translateY(6px)';
      setTimeout(() => {
        el.style.transition = 'opacity 420ms ease, transform 420ms ease';
        el.style.opacity = 1;
        el.style.transform = 'translateY(0)';
      }, 80 + i * 60);
    });
  });

})();
